9. Add Rating using $push

---

db.Books.updateOne(
{ title: "The Guide" },
{

user: "Surya",

}
}
}
)

---

10. Retrieve Books Published After 2015

---

db.Books.find({ publicationYear: { $gt: 2015 } })

---

11. Retrieve Users Joined in Last 6 Months

---

db.Users.find({
joinDate: { $gte: "2025-09-01" }
})

---

12. Books with Average Rating Greater Than 4

---

db.Books.aggregate([
{
$addFields: {
avgRating: { $avg: "$ratings.score" }
}
},
{
$match: {
avgRating: { $gt: 4 }
}
}
])
